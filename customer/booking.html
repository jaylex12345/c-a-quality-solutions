function calculateDistanceAndPrice() { const pickup = document.getElementById("pickup").value; const dropoff = document.getElementById("dropoff").value; const resultBox = document.getElementById("result"); const milesInput = document.getElementById("miles"); const priceInput = document.getElementById("estimatedPriceInput"); if (!pickup || !dropoff) { alert("Please enter pickup and drop-off address."); return; } const service = new google.maps.DistanceMatrixService(); service.getDistanceMatrix( { origins: [pickup], destinations: [dropoff], travelMode: google.maps.TravelMode.DRIVING, unitSystem: google.maps.UnitSystem.IMPERIAL }, function (response, status) { if (status !== "OK") { alert("Unable to calculate distance. Please check the addresses."); return; } const element = response.rows[0].elements[0]; if (element.status !== "OK") { alert("No route found. Please check the addresses."); return; } const miles = element.distance.value / 1609.344; const roundedMiles = miles.toFixed(1); const serviceType = document.querySelector('[name="Service Type"]').value; let basePrice = 45; let mileRate = 2.25; if (serviceType === "Priority / Same-Day Delivery") { basePrice = 70; mileRate = 2.50; } if (serviceType === "Medical Courier") { basePrice = 75; mileRate = 2.50; } if (serviceType === "Legal Courier") { basePrice = 65; mileRate = 2.25; } if (serviceType === "Property Services") { basePrice = 65; mileRate = 2.25; } if (serviceType === "Dedicated Route / Monthly Contract") { resultBox.innerText = "Custom quote required. Please submit your request."; priceInput.value = "Custom Quote"; return; } const price = basePrice + (miles * mileRate); milesInput.value = roundedMiles; priceInput.value = "$" + price.toFixed(2); resultBox.innerText = "Estimated Price: $" + price.toFixed(2) + " (" + roundedMiles + " miles)"; } ); }
